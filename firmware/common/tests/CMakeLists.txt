cmake_minimum_required(VERSION 3.16)
project(bansayo_firmware_tests C)
enable_testing()
add_executable(test_imu test_imu.c ../imu_protocol.c ../ahrs.c)
target_include_directories(test_imu PRIVATE ..)
target_compile_features(test_imu PRIVATE c_std_11)
if(NOT MSVC)
    target_compile_options(test_imu PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(test_imu PRIVATE m)
endif()
add_test(NAME firmware_imu COMMAND test_imu)
